    30	/**
    31	 * Maps KeyboardEvent.code to the modifier prefix used in combo shortcuts
    32	 * (e.g., "Command+K"). These are the generic modifier names without left/right
    33	 * distinction, used when building combo hotkey strings.
    34	 */
    35	function codeToComboModifier(code: string): string | undefined {
    36		switch (code) {
    37			case "MetaLeft":
    38			case "MetaRight":
    39				return "Command";
    40			case "ShiftLeft":
    41			case "ShiftRight":
    42				return "Shift";
    43			case "AltLeft":
    44			case "AltRight":
    45				return "Alt";
    46			case "ControlLeft":
    47			case "ControlRight":
    48				return "Control";
    49			default:
    50				return undefined;
    51		}
    52	}

    54	/**
    55	 * Consistent ordering for modifier keys when building shortcut strings.

    57	const MOD_ORDER = [
    58		"MetaLeft",
    59		"MetaRight",
    60		"ControlLeft",
    61		"ControlRight",
    62		"AltLeft",
    63		"AltRight",
    64		"ShiftLeft",

    93	interface UseHotkeyRecorderOptions {
    94		onHotkeyRecorded?: (hotkey: string) => Promise<void>;
    95		validator?: (hotkey: string) => string | null;
    96		recorderId?: string;

    99	function keyEventToHotkeyKey(e: KeyboardEvent): string {
   100		if (/^Key[A-Z]$/.test(e.code)) return e.code.slice(3);
   101		if (/^Digit[0-9]$/.test(e.code)) return e.code.slice(5);
   102		if (/^Numpad/.test(e.code)) return e.code;
   103		if (/^F[0-9]{1,2}$/.test(e.code)) return e.code;
   104		return CODE_TO_HOTKEY_KEY[e.code] ?? (e.key.length === 1 ? e.key.toUpperCase() : e.key);
   105	}

   107	export function useHotkeyRecorder(options: UseHotkeyRecorderOptions = {}) {
   108		const recorderId = options.recorderId || Math.random().toString(36);
   109		const [isRecording, setIsRecording] = createSignal(false);
   110		const [pendingHotkey, setPendingHotkey] = createSignal("");
   111		const [error, setError] = createSignal<string | null>(null);
   112	
   113		// Track all currently held modifier codes for combo shortcuts
   114		let heldModifierCodes = new Set<string>();
   115		let pendingSystemHotkey = "";
   116		let isStartingWatcher = false;
   117	
   118		const resetTracking = () => {
   119			heldModifierCodes = new Set<string>();
   120			pendingSystemHotkey = "";
   121		};
   122	
   123		const setRecorderActive = (active: boolean) => {
   124			void Result.tryPromise(() => emit("hotkey-recorder-active", { active }));
   125		};
   126	
   127		/**
   128		 * Tear down the recording session. Releases the shared `activeRecordingId`
   129		 * lock so another recorder can start — this must run on every completion
   130		 * path (successful save *and* cancel), otherwise the lock stays held and
   131		 * the next recorder is wrongly told "another hotkey is being recorded".
   132		 */
   133		const finishRecording = () => {
   134			if (activeRecordingId === recorderId) {
   135				activeRecordingId = null;
   136			}
   137			setIsRecording(false);
   138			setPendingHotkey("");
   139			resetTracking();
   140			setRecorderActive(false);
   141		};
   142	
   143		const saveRecordedHotkey = async (hotkey: string) => {
   144			const validationError = options.validator?.(hotkey);
   145			if (validationError) {
   146				setError(validationError);
   147				return false;
   148			}
   149			setError(null);
   150			const saved = await Result.tryPromise(async () => {
   151				await (options.onHotkeyRecorded?.(hotkey) ?? updateHotkey(hotkey));
   152			});
   153			if (Result.isError(saved)) {
   154				return false;
   155			}
   156			return true;
   157		};
   158	
   159		/**
   160		 * Build a display string for currently held modifiers.
   161		 */
   162		const buildModifierDisplay = (codes: Set<string>): string => {
   163			if (codes.size === 1) {
   164				const first = codes.values().next();
   165				if (first.done) return "";
   166				return modifierCodeToDisplay(first.value) ?? first.value;
   167			}
   168			const parts: string[] = [];
   169			for (const mc of MOD_ORDER) {
   170				if (codes.has(mc)) {
   171					const display = modifierCodeToDisplay(mc);
   172					if (display) parts.push(display);
   173				}
   174			}
   175			return parts.join(" + ");
   176		};
   177	
   178		const handleKeyDown = (e: KeyboardEvent) => {
   179			if (!isRecording()) return;
   180	
   181			e.preventDefault();
   182			e.stopPropagation();
   183	
   184			const code = e.code;
   185	
   186			if (isModifierCode(code)) {
   187				// A modifier key was pressed — show current modifier state
   188				heldModifierCodes.add(code);
   189				setPendingHotkey(buildModifierDisplay(heldModifierCodes));
   190			} else {
   191				// A non-modifier key was pressed — build a standard combo shortcut string
   192				const modifiers: string[] = [];
   193				const addedModifiers = new Set<string>();
   194	
   195				for (const mc of MOD_ORDER) {
   196					if (heldModifierCodes.has(mc)) {
   197						const modName = codeToComboModifier(mc);
   198						if (modName && !addedModifiers.has(modName)) {
   199							modifiers.push(modName);
   200							addedModifiers.add(modName);
   201						}
   202					}
   203				}
   204	
   205				const keyPart = keyEventToHotkeyKey(e);
   206				const hotkeyString = [...modifiers, keyPart].join("+");
   207				setPendingHotkey(hotkeyString);
   208			}
   209		};
   210	
   211		const handleKeyUp = async (e: KeyboardEvent) => {
   212			if (!isRecording()) return;
   213	
   214			const code = e.code;
   215	
   216			if (isModifierCode(code)) {
   217				const systemHotkey = modifierCodeToSystemHotkey(code);
   218				if (systemHotkey && pendingSystemHotkey) {
   219					heldModifierCodes.delete(code);
   220					return;
   221				}
   222	
   223				if (heldModifierCodes.size === 1 && heldModifierCodes.has(code) && systemHotkey) {
   224					const success = await saveRecordedHotkey(systemHotkey);
   225					if (success) {
   226						finishRecording();
   227					}
   228					return;
   229				}
   230	
   231				heldModifierCodes.delete(code);
   232	
   233				// When all modifiers are released without a combo, just reset display
   234				if (heldModifierCodes.size === 0) {
   235					setPendingHotkey("");
   236				} else {
   237					// Update display with remaining modifiers
   238					setPendingHotkey(buildModifierDisplay(heldModifierCodes));
   239				}
   240			} else {
   241				// A non-modifier key was released — check if we have a valid combo shortcut
   242				const pending = pendingHotkey();
   243				if (pending?.includes("+")) {
   244					const success = await saveRecordedHotkey(pending);
   245					if (success) {
   246						finishRecording();
   247					}
   248				}
   249			}
   250		};
   251	
   252		const unlistenSystemPressedPromise = listen<SystemKeyPressedPayload>(
   253			"system-key-pressed",
   254			(event) => {
   255				if (!isRecording()) return;
   256	
   257				pendingSystemHotkey = systemHotkeyFromKeys(event.payload.pressedKeys);
   258				setPendingHotkey(hotkeyDisplayName(pendingSystemHotkey));
   259			}
   260		);
   261	
   262		const unlistenSystemReleasedPromise = listen("system-keys-released", () => {
   263			if (!isRecording()) return;
   264			if (!pendingSystemHotkey) return;
   265	
   266			const hotkey = pendingSystemHotkey;
   267			void Result.tryPromise(() =>
   268				saveRecordedHotkey(hotkey).then((success) => {
   269					if (success) {
   270						finishRecording();
   271					}
   272				})
   273			);
   274		});
   275	
   276		createEffect(() => {
   277			if (isRecording()) {
   278				resetTracking();
   279				window.addEventListener("keydown", handleKeyDown);
   280				window.addEventListener("keyup", handleKeyUp);
   281			} else {
   282				window.removeEventListener("keydown", handleKeyDown);
   283				window.removeEventListener("keyup", handleKeyUp);
   284			}
   285		});
   286	
   287		onCleanup(() => {
   288			window.removeEventListener("keydown", handleKeyDown);
   289			window.removeEventListener("keyup", handleKeyUp);
   290			if (activeRecordingId === recorderId) {
   291				activeRecordingId = null;
   292			}
   293			setRecorderActive(false);
   294			void Result.tryPromise(() => unlistenSystemPressedPromise.then((unlisten) => unlisten()));
   295			void Result.tryPromise(() => unlistenSystemReleasedPromise.then((unlisten) => unlisten()));
   296		});
   297	
   298		const startRecording = async () => {
   299			if (activeRecordingId !== null && activeRecordingId !== recorderId) {
   300				setError("Another hotkey is being recorded. Please finish or cancel it first.");
   301				return;
   302			}
   303			if (isStartingWatcher) return;
   304			isStartingWatcher = true;
   305			const watcherStarted = await startSystemKeyWatcher();
   306			isStartingWatcher = false;
   307			if (Result.isError(watcherStarted)) {
   308				setError(watcherStarted.error.message);
   309				return;
   310			}
   311			activeRecordingId = recorderId;
   312			setIsRecording(true);
   313			setPendingHotkey("");
   314			setError(null);
   315			resetTracking();
   316			setRecorderActive(true);
   317		};
   318	
   319		const stopRecording = () => {
   320			finishRecording();
   321			setError(null);
   322		};
   323	
   324		return {
   325			isRecording,
   326			pendingHotkey,
   327			error,
   328			startRecording,
   329			stopRecording,
   330			toggleRecording: () => {
   331				if (isRecording()) {
   332					stopRecording();
   333				} else {
   334					void startRecording();
   335				}
   336			},
   337		};
   338	}

    10	const UPDATE_CHECK_INTERVAL_MS = 6 * 60 * 60 * 1000;
    11	
    12	export default function UpdateNotification() {
    13		const [t] = useI18n();
    14		const [update, setUpdate] = createSignal<Update | null>(null);
    15		const [isVisible, setIsVisible] = createSignal(false);
    16		const [isDownloading, setIsDownloading] = createSignal(false);
    17		const [downloadProgress, setDownloadProgress] = createSignal(0);
    18		let isCheckingForUpdates = false;
    19	
    20		const checkForUpdates = async () => {
    21			if (isCheckingForUpdates || isDownloading()) {
    22				logDiagnostic("debug", "updater", "check_skipped", {
    23					isCheckingForUpdates,
    24					isDownloading: isDownloading(),
    25				});
    26				return;
    27			}
    28	
    29			isCheckingForUpdates = true;
    30			const startedAt = Date.now();
    31			logDiagnostic("debug", "updater", "check_started");
    32			try {
    33				const available = await Result.tryPromise(() => check());
    34				if (Result.isOk(available) && available.value) {
    35					logDiagnostic("info", "updater", "update_available", {
    36						version: available.value.version,
    37						elapsedMs: Date.now() - startedAt,
    38					});
    39					setUpdate(available.value);
    40					setIsVisible(true);
    41				} else if (Result.isOk(available)) {
    42					logDiagnostic("debug", "updater", "no_update", {
    43						elapsedMs: Date.now() - startedAt,
    44					});
    45					setUpdate(null);
    46					setIsVisible(false);
    47				} else {
    48					logDiagnostic("error", "updater", "check_failed", {
    49						elapsedMs: Date.now() - startedAt,
    50						error: errorFields(available.error),
    51					});
    52				}
    53			} finally {
    54				isCheckingForUpdates = false;
    55			}
    56		};
    57	
    58		onMount(() => {
    59			void checkForUpdates();
    60			const intervalId = window.setInterval(() => {
    61				void checkForUpdates();
    62			}, UPDATE_CHECK_INTERVAL_MS);
    63	
    64			let cleanup: (() => void) | undefined;
    65			let disposed = false;
    66			void listen("check-for-updates", () => {
    67				void checkForUpdates();
    68			}).then((unlisten) => {
    69				if (disposed) {
    70					unlisten();
    71				} else {
    72					cleanup = unlisten;
    73				}
    74			});
    75	
    76			onCleanup(() => {
    77				disposed = true;
    78				window.clearInterval(intervalId);
    79				cleanup?.();
    80			});
    81		});
    82	
    83		const handleDownloadAndRestart = async () => {
    84			const updateInfo = update();
    85			if (!updateInfo) return;
    86	
    87			logDiagnostic("info", "updater", "download_and_restart_requested", {
    88				version: updateInfo.version,
    89			});
    90			setIsDownloading(true);
    91			let downloadedBytes = 0;
    92			const installed = await Result.tryPromise({
    93				try: async () => {
    94					await updateInfo.downloadAndInstall((event) => {
    95						if (event.event === "Started" && event.data.contentLength) {
    96							downloadedBytes = 0;
    97							logDiagnostic("info", "updater", "download_started", {
    98								version: updateInfo.version,
    99								contentLength: event.data.contentLength,
   100							});
   101							setDownloadProgress(0);
   102						} else if (event.event === "Progress") {
   103							downloadedBytes += event.data.chunkLength;
   104							const progress = downloadProgress() + event.data.chunkLength;
   105							setDownloadProgress(progress);
   106						} else if (event.event === "Finished") {
   107							logDiagnostic("info", "updater", "download_finished", {
   108								version: updateInfo.version,
   109								downloadedBytes,
   110							});
   111							setDownloadProgress(100);
   112						}
   113					});
   114					logDiagnostic("warn", "updater", "relaunch_requested", {
   115						version: updateInfo.version,
   116					});
   117					await relaunch();
   118				},
   119				catch: (cause) => cause,
   120			});
   121			if (Result.isError(installed)) {
   122				logDiagnostic("error", "updater", "download_or_install_failed", {
   123					version: updateInfo.version,
   124					error: errorFields(installed.error),
   125				});
   126				setIsDownloading(false);
   127			}
   128		};
   129	
   130		return (
   131			<Show when={isVisible() && update()}>
   132				<div class="border-t border-border">
   133					<div class="px-6 py-3">
   134						<div class="mb-1.5">
   135							<span class="font-mono uppercase tracking-wider text-[10px] text-ac flex items-center gap-1.5">
   136								<Download class="h-2.5 w-2.5" />
   137								{t("update.available")}
   138							</span>
   139						</div>
   140	
   141						<div class="font-mono text-xs tabular-nums mb-2">
   142							<span class="text-txt-primary">v{update()?.version}</span>
   143						</div>
   144	
   145						<Show
   146							when={!isDownloading()}
   147							fallback={
   148								<>
   149									<div class="w-full h-1 bg-border overflow-hidden">
   150										<div
   151											class="h-full bg-ac transition-all duration-300"
   152											style={{ width: `${Math.min(downloadProgress() / 10, 100)}%` }}
   153										/>
   154									</div>
   155									<p class="mt-1.5 font-mono uppercase tracking-wider text-[10px] text-txt-muted">
   156										{t("update.downloading")}
   157									</p>
   158								</>
   159							}
   160						>
   161							<button
   162								type="button"
   163								onClick={handleDownloadAndRestart}
   164								class="w-full flex items-center justify-center gap-1.5 px-2 py-1.5 font-mono uppercase tracking-wider text-[10px] border border-ac text-ac hover:bg-ac hover:text-ac-on transition-colors"
   165							>
   166								<span aria-hidden="true">→</span>
   167								{t("update.downloadAndRestart")}
   168							</button>
   169						</Show>
   170					</div>
   171				</div>
   172			</Show>
   173		);
   174	}
