     3	import type { SelectOption } from "./types";
     4	
     5	interface SelectProps {
     6		value: string;
     7		options: SelectOption[];
     8		onChange: (value: string) => void;
     9		placeholder?: string;
    10	}
    11	
    12	export default function Select(props: SelectProps) {
    13		const [isOpen, setIsOpen] = createSignal(false);
    14		let containerRef: HTMLDivElement | undefined;
    15	
    16		const selectedLabel = () => {
    17			const option = props.options.find((o) => o.value === props.value);
    18			return option?.label ?? props.placeholder ?? "";
    19		};
    20	
    21		const handleClickOutside = (e: MouseEvent) => {
    22			if (containerRef && !containerRef.contains(e.target as Node)) {
    23				setIsOpen(false);
    24			}
    25		};
    26	
    27		createEffect(() => {
    28			if (isOpen()) {
    29				document.addEventListener("click", handleClickOutside);
    30			} else {
    31				document.removeEventListener("click", handleClickOutside);
    32			}
    33		});
    34	
    35		onCleanup(() => {
    36			document.removeEventListener("click", handleClickOutside);
    37		});
    38	
    39		return (
    40			<div ref={containerRef} class="relative">
    41				<button
    42					type="button"
    43					onClick={() => setIsOpen(!isOpen())}
    44					class="flex items-center justify-between w-full px-4 py-2.5 bg-th-surface border border-border-strong text-txt-primary hover:border-ac focus:outline-none focus:border-ac transition-colors font-mono text-sm"

     1	import { Show, createSignal } from "solid-js";
     2	import { useI18n } from "../../i18n";
     3	import { ONBOARDING_STEP_COUNT } from "../../lib/onboarding";
     4	import { capture } from "../../lib/posthog";
     5	import { updateOnboardingStep } from "../../lib/settingsStore";
     6	import StepIndicator from "./StepIndicator";
     7	import AccessibilityPermissionStep from "./steps/AccessibilityPermissionStep";
     8	import CompletionStep from "./steps/CompletionStep";
     9	import HotkeyStep from "./steps/HotkeyStep";
    10	import LearningStep from "./steps/LearningStep";
    11	import MicrophonePermissionStep from "./steps/MicrophonePermissionStep";
    12	import MicrophoneStep from "./steps/MicrophoneStep";
    13	import ModelDownloadStep from "./steps/ModelDownloadStep";
    14	
    15	interface OnboardingWizardProps {
    16		initialStep: number;
    17		onComplete: () => void;
    18	}
    19	
    20	export default function OnboardingWizard(props: OnboardingWizardProps) {
    21		const [t] = useI18n();
    22		const [currentStep, setCurrentStep] = createSignal(props.initialStep);
    23		const [isAnimating, setIsAnimating] = createSignal(false);
    24		const [animationDirection, setAnimationDirection] = createSignal<"forward" | "backward">(
    25			"forward"
    26		);
    27		const [micPermissionGranted, setMicPermissionGranted] = createSignal(false);
    28		const [accessibilityPermissionGranted, setAccessibilityPermissionGranted] = createSignal(false);
    29		const [modelDownloaded, setModelDownloaded] = createSignal(false);
    30		const [learningCompleted, setLearningCompleted] = createSignal(false);
    31	
    32		const goToNext = () => {
    33			if (currentStep() < ONBOARDING_STEP_COUNT) {
    34				capture("onboarding_step_completed", { step: currentStep() });
    35				setAnimationDirection("forward");
    36				setIsAnimating(true);
    37				setTimeout(() => {
    38					const nextStep = currentStep() + 1;
    39					setCurrentStep(nextStep);
    40					setIsAnimating(false);
    41					updateOnboardingStep(nextStep);
    42				}, 200);
    43			}
    44		};
    45	
    46		const goToPrevious = () => {
    47			if (currentStep() > 1) {
    48				setAnimationDirection("backward");
    49				setIsAnimating(true);
    50				setTimeout(() => {
    51					const prevStep = currentStep() - 1;
    52					setCurrentStep(prevStep);
    53					setIsAnimating(false);
    54					updateOnboardingStep(prevStep);
    55				}, 200);
    56			}
    57		};
    58	
    59		const handleComplete = () => {
    60			props.onComplete();
    61		};
    62	
    63		const canProceed = () => {
    64			const step = currentStep();
    65			if (step === 1) return micPermissionGranted();
    66			if (step === 2) return accessibilityPermissionGranted();
    67			if (step === 5) return modelDownloaded();
    68			if (step === 6) return learningCompleted();
    69			return true;
    70		};
    71	
    72		const formatStep = (step: number) => step.toString().padStart(2, "0");
    73	
    74		return (
    75			<div class="fixed inset-0 bg-th-base flex flex-col items-center justify-center p-8">
    76				<div
    77					class="absolute inset-0 pointer-events-none"
    78					style={{
    79						"background-image":
    80							"linear-gradient(var(--color-grid-line) 1px, transparent 1px), linear-gradient(90deg, var(--color-grid-line) 1px, transparent 1px)",
    81						"background-size": "40px 40px",
    82					}}
    83				/>
    84	
    85				<div class="absolute top-0 left-0 right-0 h-6 z-50" data-tauri-drag-region />
    86	
    87				<div class="absolute top-8 right-8 font-mono text-txt-muted text-sm tracking-wider">
    88					{formatStep(currentStep())}/{formatStep(ONBOARDING_STEP_COUNT)}
    89				</div>
    90	
    91				<div class="mb-12">
    92					<StepIndicator currentStep={currentStep()} totalSteps={ONBOARDING_STEP_COUNT} />
    93				</div>
    94	
    95				<div
    96					class={`flex-1 flex items-center justify-center w-full transition-all duration-300 ${currentStep() === 6 ? "max-w-4xl" : "max-w-2xl"}`}
    97				>
    98					<div
    99						class={`w-full transition-all duration-200 ${
   100							isAnimating()
   101								? animationDirection() === "forward"
   102									? "opacity-0 translate-x-8"
   103									: "opacity-0 -translate-x-8"
   104								: "opacity-100 translate-x-0"
   105						}`}
   106					>
   107						<Show when={currentStep() === 1}>
   108							<MicrophonePermissionStep onPermissionChange={setMicPermissionGranted} />
   109						</Show>
   110						<Show when={currentStep() === 2}>
   111							<AccessibilityPermissionStep onPermissionChange={setAccessibilityPermissionGranted} />
   112						</Show>
   113						<Show when={currentStep() === 3}>
   114							<MicrophoneStep />
   115						</Show>
   116						<Show when={currentStep() === 4}>
   117							<HotkeyStep />
   118						</Show>
   119						<Show when={currentStep() === 5}>
   120							<ModelDownloadStep onDownloadComplete={() => setModelDownloaded(true)} />
   121						</Show>
   122						<Show when={currentStep() === 6}>
   123							<LearningStep onTranscriptionComplete={() => setLearningCompleted(true)} />
   124						</Show>
   125						<Show when={currentStep() === 7}>
   126							<CompletionStep />
   127						</Show>
   128					</div>
   129				</div>
   130	
   131				<div
   132					class={`flex items-center justify-between w-full mt-8 transition-all duration-300 ${currentStep() === 6 ? "max-w-4xl" : "max-w-2xl"}`}
   133				>
   134					<Show when={currentStep() > 1} fallback={<div />}>
   135						<button
   136							type="button"
   137							onClick={goToPrevious}
   138							class="px-6 py-3 font-mono text-txt-secondary hover:text-txt-primary transition-colors uppercase tracking-wider text-sm border border-border-strong hover:border-ac"
   139						>
   140							{t("onboarding.back")}
   141						</button>

   150								class="px-8 py-3 bg-ac text-ac-on font-mono font-bold uppercase tracking-wider text-sm hover:bg-ac-hover transition-colors"
   151							>
   152								{t("onboarding.getStarted")}
   153							</button>
   154						}
   155					>
   156						<button
   157							type="button"
   158							onClick={goToNext}
   159							disabled={!canProceed()}
   160							class="px-8 py-3 bg-ac text-ac-on font-mono font-bold uppercase tracking-wider text-sm hover:bg-ac-hover transition-colors disabled:opacity-30"
   161						>
