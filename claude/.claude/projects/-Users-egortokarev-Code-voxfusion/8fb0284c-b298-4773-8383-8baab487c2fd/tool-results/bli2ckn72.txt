     1	import {
     2		debug as logDebug,
     3		error as logError,
     4		info as logInfo,
     5		warn as logWarn,
     6	} from "@tauri-apps/plugin-log";
     7	
     8	type DiagnosticLevel = "debug" | "info" | "warn" | "error";
     9	type DiagnosticFields = Record<string, unknown>;
    10	
    11	export function logDiagnostic(
    12		level: DiagnosticLevel,
    13		target: string,
    14		message: string,
    15		fields: DiagnosticFields = {}
    16	): void {
    17		const logMessage = `${target}.${message}`;
    18		const options = { keyValues: stringifyFields(fields) };
    19		const logger = {
    20			debug: logDebug,
    21			info: logInfo,
    22			warn: logWarn,
    23			error: logError,
    24		}[level];
    25	
    26		void logger(logMessage, options).catch(() => {
    27			// Logging must never become part of the app's control flow.
    28		});
    29	}
    30	
    31	export function errorFields(cause: unknown): DiagnosticFields {
    32		if (cause instanceof Error) {
    33			return {
    34				name: cause.name,
    35				message: cause.message,
    36				stack: cause.stack,
    37			};
    38		}
    39	
    40		return {
    41			message: String(cause),
    42		};
    43	}
    44	
    45	function stringifyFields(fields: DiagnosticFields): Record<string, string | undefined> {
    46		return Object.fromEntries(
    47			Object.entries(fields).map(([key, value]) => [
    48				key,
    49				value === undefined ? undefined : serializeField(value),
    50			])
    51		);
    52	}
    53	
    54	function serializeField(value: unknown): string {
    55		if (typeof value === "string") return value;
    56	
    57		try {
    58			return JSON.stringify(value);
    59		} catch {
    60			return String(value);
    61		}
    62	}
