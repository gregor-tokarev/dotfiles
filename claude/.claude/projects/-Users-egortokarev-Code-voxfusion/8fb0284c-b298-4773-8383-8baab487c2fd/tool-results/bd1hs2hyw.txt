    20		const markGranted = () => {
    21			setIsGranted(true);
    22			props.onPermissionChange(true);
    23			if (pollInterval) {
    24				clearInterval(pollInterval);
    25				pollInterval = undefined;
    26			}
    27		};
    28	
    29		const checkPermission = async () => {
    30			const granted = await checkAccessibilityProbe();
    31			if (Result.isError(granted)) {
    32				setIsGranted(false);
    33				props.onPermissionChange(false);
    34				return;
    35			}
    36			setIsGranted(granted.value);
    37			props.onPermissionChange(granted.value);
    38			if (granted.value && pollInterval) {
    39				clearInterval(pollInterval);
    40				pollInterval = undefined;
    41			}
    42		};
    43	
    44		const checkWithRetries = async () => {
    45			// Try immediately
    46			await checkPermission();
    47			if (isGranted()) return;
    48	
    49			// Retry after 200ms (let TCC settle)
    50			await new Promise((r) => setTimeout(r, 200));
    51			await checkPermission();
    52			if (isGranted()) return;
    53	
    54			// Retry after 500ms more

    57			if (isGranted()) return;

    60			// trust the notification since it only fires on an actual toggle
    61			await new Promise((r) => setTimeout(r, 1000));
